const $ = (sel) => document.querySelector(sel);

function clamp(n, a, b){ return Math.max(a, Math.min(b, n)); }
function round2(n){ return Math.round(n * 100) / 100; }

function impliedProbFromOdds(odds){
  if(!odds || odds <= 1) return null;
  return 1 / odds;
}

function fmtPct(p){
  if(p === null || p === undefined) return "—";
  return `${Math.round(p * 100)}%`;
}

function riskPillClass(risk){
  const r = (risk || "").toLowerCase();
  if (r.includes("faible")) return "pill--good";
  if (r.includes("mod")) return "pill--warn";
  if (r.includes("élev")) return "pill--bad";
  return "";
}

function setConfidence(score){
  const s = clamp(Number(score) || 0, 0, 100);
  $("#confidenceValue").textContent = String(s);
  $("#confidenceBar").style.width = `${s}%`;
}

function renderBetsTable(bets){
  const tbody = $("#betsTable tbody");
  if(!bets || bets.length === 0){
    tbody.innerHTML = `<tr><td colspan="6" class="muted">Aucun pari.</td></tr>`;
    return;
  }

  // Calcul implied prob + écart
  const normalized = bets.map(b => {
    const implied = (b.impliedProb ?? impliedProbFromOdds(b.odds));
    const est = b.estimatedProb ?? null;
    const gap = (implied !== null && est !== null) ? (est - implied) : null;
    return { ...b, impliedProb: implied, gap };
  });

  // Tri : du plus sûr au plus risqué (faible -> modéré -> élevé) puis gap desc
  const riskRank = (r="") => {
    const x = r.toLowerCase();
    if(x.includes("faible")) return 0;
    if(x.includes("mod")) return 1;
    if(x.includes("élev")) return 2;
    return 3;
  };

  normalized.sort((a,b) => {
    const rr = riskRank(a.risk) - riskRank(b.risk);
    if(rr !== 0) return rr;
    return (b.gap ?? -999) - (a.gap ?? -999);
  });

  tbody.innerHTML = normalized.map(b => `
    <tr>
      <td>
        <div style="font-weight:800">${b.bet}</div>
        <div class="muted" style="font-size:12px; margin-top:4px">${b.why || ""}</div>
      </td>
      <td>${b.odds ?? "—"}</td>
      <td>${fmtPct(b.impliedProb)}</td>
      <td>${fmtPct(b.estimatedProb)}</td>
      <td>${b.gap === null ? "—" : `${round2(b.gap*100)} pts`}</td>
      <td><span class="pill ${riskPillClass(b.risk)}">${b.risk || "—"}</span></td>
    </tr>
  `).join("");
}

function renderBias(items){
  const ul = $("#biasList");
  if(!items || items.length === 0){
    ul.innerHTML = `<li class="muted">Aucun biais.</li>`;
    return;
  }
  ul.innerHTML = items.map(x => `
    <li>
      <div style="display:flex; justify-content:space-between; gap:10px">
        <div style="font-weight:800">${x.name}</div>
        <div class="muted">${x.status}</div>
      </div>
      <div class="muted" style="font-size:12px; margin-top:6px">${x.note || ""}</div>
    </li>
  `).join("");
}

function renderQuestions(qs){
  const ul = $("#questionsList");
  if(!qs || qs.length === 0){
    ul.innerHTML = `<li class="muted">Aucune question.</li>`;
    return;
  }
  ul.innerHTML = qs.map(q => `<li>${q}</li>`).join("");
}

function readForm(){
  return {
    sport: $("#sport").value,
    competition: $("#competition").value.trim(),
    date: $("#date").value,
    teamA: $("#teamA").value.trim(),
    teamB: $("#teamB").value.trim(),
    market: $("#market").value
  };
}

function mergeUserMatchIntoSample(form){
  const sample = structuredClone(window.SMALTOVISI_SAMPLE);
  sample.match.sport = form.sport || sample.match.sport;
  sample.match.competition = form.competition || sample.match.competition;
  sample.match.date = form.date || sample.match.date;
  sample.match.teamA = form.teamA || sample.match.teamA;
  sample.match.teamB = form.teamB || sample.match.teamB;
  return sample;
}

function init(){
  $("#year").textContent = String(new Date().getFullYear());

  $("#loadDemo").addEventListener("click", () => {
    $("#sport").value = "football";
    $("#competition").value = window.SMALTOVISI_SAMPLE.match.competition;
    $("#date").value = window.SMALTOVISI_SAMPLE.match.date;
    $("#teamA").value = window.SMALTOVISI_SAMPLE.match.teamA;
    $("#teamB").value = window.SMALTOVISI_SAMPLE.match.teamB;
    $("#market").value = "1X2";
    buildReport(window.SMALTOVISI_SAMPLE);
  });

  $("#matchForm").addEventListener("submit", (e) => {
    e.preventDefault();
    const form = readForm();
    const payload = mergeUserMatchIntoSample(form);
    buildReport(payload);
    document.getElementById("reportCard").scrollIntoView({ behavior: "smooth", block: "start" });
  });
}

init();
